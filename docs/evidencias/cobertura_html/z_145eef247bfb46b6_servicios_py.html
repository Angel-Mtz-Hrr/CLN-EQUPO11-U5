1"""Servicios de aplicacion: coordinan reglas (dominio), base de datos y avisos/cobros externos. 

2 

3Idea clave para entender este archivo, sin tecnicismos: 

4 

5 1. PRIMERO se decide y se GUARDA lo importante (la reserva, la cancelacion). 

6 2. DESPUES se avisa al socio. Un aviso que falla NO deshace una reserva ya confirmada: 

7 es mejor que el socio tenga su lugar y no reciba el correo, a que pierda su lugar 

8 porque el correo estaba caido. 

9 3. Un COBRO es distinto: si el banco rechaza, no hay nada que avisar como exito. 

10 

11Los servicios externos (Notificador, PasarelaDePago) llegan "de afuera": en produccion son 

12reales; en las pruebas son dobles (mocks) que no envian ni cobran nada. 

13"""

71class ServicioDeReservas: 

72 def __init__(self, uow: UnitOfWork, notificador: Notificador): 

73 self._uow = uow 

74 self._notificador = notificador 

75 

76 def reservar( 

77 self, clase_id: str, socio_id: str, actor: Actor, ahora: datetime 

78 ) -> ResultadoDeReserva: 

79 # Primero el permiso (BR-03): a quien no puede, ni siquiera se le consulta la base. 

80 if not actor.puede_gestionar_reserva_de(socio_id): 

81 raise PermisoDenegado("este rol no puede reservar en nombre de ese socio") 

82 

83 with self._uow as uow: 

84 clase = uow.clases.obtener_por_id(clase_id) 

85 if clase is None: 

86 raise ClaseNoEncontrada(f"no existe la clase {clase_id}") 

87 membresia = uow.membresias.obtener_actual_del_socio(socio_id) 

88 if membresia is None: 

89 raise MembresiaNoVigente("el socio no tiene membresia") 

90 plan = uow.planes.obtener_por_id(membresia.plan_id) 

91 ReglasDeElegibilidad().evaluar(membresia, plan, clase, ahora).exigir() # BR-06, BR-07 

92 reserva = clase.reservar(socio_id, actor, ahora) # BR-05 (cupo) 

93 uow.clases.guardar(clase) 

94 uow.confirmar() 

95 

96 aviso = _avisar( 

97 self._notificador, 

98 socio_id, 

99 f"Tu lugar en la clase de {clase.tipo} ({clase.clase_id}) quedo confirmado.", 

100 ) 

101 return ResultadoDeReserva(reserva.reserva_id, aviso) 

102 

103 def cancelar_clase(self, clase_id: str, actor: Actor) -> ResultadoDeCancelacion: 

104 if not actor.puede_administrar_clases(): # BR-01 

105 raise PermisoDenegado("solo el Administrador puede cancelar una clase") 

106 

107 with self._uow as uow: 

108 clase = uow.clases.obtener_por_id(clase_id) 

109 if clase is None: 

110 raise ClaseNoEncontrada(f"no existe la clase {clase_id}") 

111 afectados = [r.socio_id for r in clase.reservas if r.estado is EstadoDeReserva.ACTIVA] 

112 clase.cancelar(actor) 

113 uow.clases.guardar(clase) 

114 uow.confirmar() 

115 

116 avisados, sin_aviso = [], [] 

117 for socio_id in afectados: # un aviso fallido no impide avisar a los demas 

118 ok = _avisar( 

119 self._notificador, 

120 socio_id, 

121 f"Lamentamos avisarte que la clase de {clase.tipo} ({clase.clase_id}) se cancelo.", 

122 ) 

123 (avisados if ok else sin_aviso).append(socio_id) 

124 return ResultadoDeCancelacion(tuple(avisados), tuple(sin_aviso))

127class ServicioDePagos: 

128 """Cobros a traves de la pasarela de pago (en pruebas: un doble que NO cobra).""" 

129 

130 def __init__(self, pasarela: PasarelaDePago, notificador: Notificador): 

131 self._pasarela = pasarela 

132 self._notificador = notificador 

133 

134 def cobrar( 

135 self, socio_id: str, monto_centavos: int, concepto: str, actor: Actor 

136 ) -> ResultadoDeCobro: 

137 if not actor.puede_gestionar_pago_de(socio_id): 

138 raise PermisoDenegado("este rol no puede cobrar a ese socio") 

139 if monto_centavos <= 0: 

140 raise MontoInvalido("el monto debe ser mayor que cero") 

141 

142 # Un solo intento, sin reintentos automaticos: asi jamas se cobra dos veces por error. 

143 # Si la pasarela rechaza (PagoRechazado) o no responde, el error sube tal cual y 

144 # NO se manda ningun aviso de "pago exitoso". 

145 comprobante = self._pasarela.cobrar(socio_id, monto_centavos, concepto) 

146 

147 mensaje = f"Recibimos tu pago por {concepto}. Comprobante {comprobante}." 

148 aviso = _avisar(self._notificador, socio_id, mensaje) 

149 return ResultadoDeCobro(comprobante, aviso)
